# (onnxruntime, espeak-ng) conflicts with the main venv.

# Setup instructions:
#   conda create -n kokoro python=3.10
#   conda activate kokoro
#   pip install kokoro soundfile numpy
#   brew install espeak-ng      # macOS
#   # or: apt-get install espeak-ng  (Linux)
#
# The KOKORO_PYTHON env var tells generate_audio.py where to find this env:
#   export KOKORO_PYTHON=~/miniconda3/envs/kokoro/bin/python
#
# If KOKORO_PYTHON is not set, it defaults to:
#   ~/miniconda3/envs/kokoro/bin/python
